How to Auto-Publish Articles

This website is pre-configured to automatically recognize and render any HTML files saved in the /posts/ folder, provided they are declared in your sitemap.xml.

publisher.py 
# publisher.py - Automated static article generator
import os
import datetime
import xml.etree.ElementTree as ET

# 1. Define meta attributes for the automated article
slug = "automated-article-example"
title = "How Programmatic Scripts Scale Affiliate Income"
description = "Learn the backend mechanics of programmatic content production."
category = "Strategy"
reading_time = "5 min read"
cover_image = "https://images.unsplash.com/photo-1518186285589-2f7649de83e0?w=800"
publish_date = datetime.date.today().isoformat()

# 2. Complete HTML post template
html_template = f"""<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>{title} - Static SEO Blog</title>
  <meta name="description" content="{description}">
  <meta name="publish-date" content="{publish_date}">
  <meta name="category" content="{category}">
  <meta name="reading-time" content="{reading_time}">
  <meta name="cover-image" content="{cover_image}">
  <link rel="stylesheet" href="../index.css">
  <script src="https://unpkg.com/@tailwindcss/browser@4"></script>
</head>
<body class="bg-[#faf9f6] text-stone-800">
  <!-- Content body generated -->
</body>
</html>"""

# 3. Write HTML file into /posts/
file_path = f"posts/{slug}.html"
with open(file_path, "w") as f:
    f.write(html_template)
print(f"Article written: {file_path}")

# 4. Automatically update sitemap.xml in root
sitemap_path = "sitemap.xml"
tree = ET.parse(sitemap_path)
root = tree.getroot()

# Append new url element
url_el = ET.SubElement(root, "url")
loc_el = ET.SubElement(url_el, "loc")
loc_el.text = f"https://nicheblog.example.com/posts/{slug}.html"
lastmod_el = ET.SubElement(url_el, "lastmod")
lastmod_el.text = publish_date

tree.write(sitemap_path, encoding="UTF-8", xml_declaration=True)
print("sitemap.xml successfully updated!")
